\begin{frame}{Demo na scenie: Anna na wózku szuka kawiarni w~centrum}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.6\textwidth}
      \footnotesize
      \begin{enumerate}
        \item \textbf{Potrzeby.} Profil „Wózek”: mapa i~listy pokazują tylko istotne bariery,
              miejsca sortują się według dopasowania.
        \item \textbf{Pytanie.} „Kawiarnia w~centrum, wjadę na wózku, przyda się toaleta” $\rightarrow$
              ranking z~uzasadnieniem i~brakami (np.\ brak danych: szerokie drzwi).
        \item \textbf{Karta miejsca.} Bez schodów: \alert{Nie} \textperiodcentered\ OpenStreetMap \textperiodcentered\
              2.10.2026 \textperiodcentered\ „Dane z~oficjalnego źródła”; szerokie drzwi: \textbf{Brak danych}
              $\rightarrow$ „Uzupełnij” lub pytanie do właściciela.
        \item \textbf{Po drodze.} Zgłoszenie „Brak windy” na przystanku: pinezka, typ, waga; inni głosują
              „nadal jest” / „naprawione”; awaria wygasa po 14~dniach.
        \item \textbf{Właściciel.} W~panelu uzupełnia drzwi i~potwierdza $\rightarrow$
              „Potwierdzone przez właściciela”; Anna dostaje powiadomienie.
      \end{enumerate}
      \vspace{0.5mm}
      \muted{Dane przykładowe są oznaczone: konta demo (\texttt{anna@}, \texttt{kasia@}, \texttt{admin@accessly.test}),
        zgłoszenia testowe, „Tryb demo” w~widoku administratora.}
    \end{column}
    \begin{column}{0.38\textwidth}
      \centering
      \screenshoth[36mm]{asystent-panel}\hfill\screenshoth[36mm]{karta-panel}
      \vspace{0.3mm}
      \muted{Po lewej: rekomendacje z~uzasadnieniem i~brakami. Po prawej: karta miejsca -- „Spełnia 0~z~3 potrzeb”,
        atrybut ze źródłem, datą i~statusem.}
    \end{column}
  \end{columns}

  \note[item]{Czas: ok. 3 minuty. Przed wejściem na scenę: aplikacja otwarta na telefonie i~laptopie, profil „Wózek” zaznaczony, mapa na Rynku (zoom 16).}
  \note[item]{Krok 2: wpisać zapytanie z~przykładu pod polem – wynik jest deterministyczny w~trybie reguł. Odpowiedź: „Spełnia: wejście bez schodów, toaleta; brak danych: szerokie drzwi”.}
  \note[item]{Krok 3: pokazać palcem trzy rzeczy przy atrybucie: źródło, datę, status. Potem „Brak danych” – to jest nasz najważniejszy komunikat: nie zgadujemy.}
  \note[item]{Krok 4: zgłoszenie bariery można zrobić bez konta; pokazać kształt znacznika (kwadrat = blokuje).}
  \note[item]{Krok 5: zalogować się jako \texttt{kasia@accessly.test} (konto demo właścicielki Camelot Cafe) w~/owner.html i~kliknąć „Potwierdź informacje”. Wrócić do karty – status się zmienił.}
  \note[item]{Jeśli padnie sieć: wszystko poza kafelkami mapy działa z~danych lokalnych; pokazać stronę „Źródła danych” z~datami snapshotów.}
\end{frame}
