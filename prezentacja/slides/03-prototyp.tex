      \begin{itemize}
        \item \textbf{Katalog 6\,545 miejsc} (OpenStreetMap), 7~kategorii, \textbf{27 atrybutów}: schody, podjazd, winda,
              drzwi $\geq$~90~cm, toaleta, przewijak, pętla indukcyjna, odpoczynek, parking OzN, cisza, pies\ldots
        \item \textbf{Profil potrzeb} zamiast pytania o~niepełnosprawność: filtruje, sortuje, oznacza
              „Dopasowane do Ciebie” i~mówi, czego brakuje.
        \item \textbf{Karta miejsca:} każdy fakt ze źródłem, datą i~statusem; Aktualne / Nieaktualne / Popraw / Uzupełnij,
              pytanie do właściciela, opinie, zdjęcia, trasa dla wózka.
        \item \textbf{Mapa barier} („Yanosik dostępności”): 10 typów zgłoszeń, waga blokuje / utrudnienie,
              głosy „nadal jest” / „naprawione”, automatyczne wygasanie.
        \item \textbf{Asystent w~języku naturalnym} (Claude z~wyszukiwaniem w~bazie; bez klucza -- reguły):
              pokazuje wyłącznie fakty z~bazy.
        \item \textbf{11 warstw otwartych danych}, panel właściciela, panel administratora, 4~języki, tryb nocny.
      \end{itemize}
