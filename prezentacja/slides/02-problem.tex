\begin{frame}{„Dostępne / niedostępne” to za mało}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      \begin{block}{Problem}
        \begin{itemize}
          \item Jedna etykieta nie mówi, czy wjadę z~wózkiem, czy drzwi mają 90~cm, czy jest toaleta i~gdzie odpocząć.
          \item Informacje w~sieci nie mają źródła ani daty: zepsuta winda zamienia „dostępne” w~zmarnowaną wyprawę.
          \item Brak informacji bywa pokazywany jak jej potwierdzenie, a~Miasto nie ma zasobów na ręczne utrzymanie bazy.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{Dla kogo -- zakres prototypu}
        \begin{itemize}
          \item \textbf{Grupa główna:} osoby na wózkach i~rodzice z~wózkami dziecięcymi (profile „Wózek”, „Wózek dziecięcy”).
          \item Te same dane służą profilom: Słabo widzę, Senior, Po zabiegu, Bagaż, Podróżuję z~psem, Rower.
          \item Mieszkańcy i~turyści (PL, EN, DE, UK) oraz \textbf{właściciele i~zarządcy obiektów} z~własnym panelem.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{1.5mm}
  \begin{exampleblock}{Nasza odpowiedź}
    Accessly odpowiada na pytanie \textbf{„czy to miejsce zadziała dla mnie?”} faktami o~barierach i~udogodnieniach.
    Każdy fakt ma \textbf{źródło, datę i~status weryfikacji}, a~brak danych jest brakiem danych -- nigdy dostępnością.
  \end{exampleblock}

  \note[item]{Zacząć od historii: Anna na wózku chce wyjść na kawę w~centrum. Mapa z~zeszłego roku mówi „dostępne”, a~dziś przy wejściu jest próg i~nikt nie wie, czy toaleta jest na dole.}
  \note[item]{Podkreślić dwie grupy: użytkownicy końcowi oraz właściciele – bez nich dane nie będą aktualne.}
  \note[item]{Profil to preferencje (schody, winda, cisza), nie informacja o~niepełnosprawności – wraca na slajdzie o~prywatności.}
\end{frame}
