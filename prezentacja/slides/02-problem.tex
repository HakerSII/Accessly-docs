% Slajd 2 – problem i grupa docelowa (kryterium: związek z wyzwaniem i użyteczność, 25 %).
\begin{frame}{Rzetelna informacja zamiast niepewności}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.49\textwidth}
      \begin{block}{Problem}
        \begin{itemize}
          \item Dane o~dostępności często są rozproszone, niepełne i~szybko się dezaktualizują.
          \item Bez źródła i~daty aktualizacji użytkownik nie wie, czy może tym danym zaufać.
          \item Ręczne utrzymywanie kompletnej bazy dla całego miasta może stwarzać problemy ze skalowalnością.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.49\textwidth}
      \begin{block}{Dla kogo}
        Osoby z~potrzebami dostępowymi i~ich opiekunowie, rodzice z~wózkami, seniorzy, osoby po zabiegach,
        turyści z~bagażem lub zwierzętami oraz branża turystyki medycznej.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{1mm}
  \begin{exampleblock}{Accessly}
    Łączy dane publiczne, informacje od właścicieli i~potwierdzenia społeczności.\\
    Każda informacja ma źródło, datę i~status weryfikacji.
  \end{exampleblock}

  \note[item]{Zacząć od historii: Anna na wózku chce wyjść na kawę w~centrum. Mapa z~zeszłego roku mówi „dostępne”, a~dziś przy wejściu jest próg i~nikt nie wie, czy toaleta jest na dole.}
  \note[item]{Grupa główna z~wyzwania: wózki i~wózki dziecięce (profile „Wózek”, „Wózek dziecięcy”). Pozostałe profile korzystają z~tych samych atrybutów, więc nic nie kosztują.}
  \note[item]{Podkreślić dwie grupy: użytkownicy końcowi oraz właściciele – bez nich dane nie będą aktualne.}
  \note[item]{Profil to preferencje (schody, winda, cisza), nie informacja o~niepełnosprawności – wraca na slajdzie o~prywatności.}
\end{frame}
